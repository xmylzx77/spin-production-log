// ---------- Today ----------
// Today() in a named formula is only re-read when the formula recalculates.
// Multiplying by gblRefreshTick ties it to the refresh timers, so a tablet or
// the TV left running past midnight moves to the new day on its next refresh.
TodayDate = DateAdd(Today(), 0 * Coalesce(gblRefreshTick, 0), TimeUnit.Days);
TodayYmd = Text(TodayDate, "yyyy-mm-dd");
TodayKey = Year(TodayDate) * 10000 + Month(TodayDate) * 100 + Day(TodayDate);

// ---------- Fixed lists ----------
GaugeList = [12, 14, 16, 18, 20, 24];
MachineList = ["SB8", "SB15"];
LineList = ["Line 1", "Line 2", "Line 3"];

// Today and the next 5 days (the punch board's fixed window).
PunchWindow = ForAll(
    Sequence(6, 0) As s,
    With({d: DateAdd(TodayDate, s.Value, TimeUnit.Days)},
        {BandYmd: Text(d, "yyyy-mm-dd"), BandDate: d, BandKey: Year(d) * 10000 + Month(d) * 100 + Day(d), Label: Text(d, "ddd mmm d", "en-US"), IsToday: Text(d, "yyyy-mm-dd") = TodayYmd}
    )
);

// ---------- Working set ----------

            LineText, If(IsBlank(AssemblyLine.Value), "", AssemblyLine.Value),
            MachineOrder, Switch(Machine.Value & "", "SB8", 1, "SB15", 2, 3),

// ActiveBase: one indexed query (JobStatus = Active) + the same per-row columns,

ActiveBase =

    AddColumns(
        AddColumns(

                JobSize, PairNo, PunchOrder, AssemblyOrder,
                ShipDate, PrevShipDate, PunchDay, AssemblyDate, StartedOn, Created,
                JobStatus, Machine, AssemblyLine,

                PBDone, P4Done, Started

            LineText, If(IsBlank(AssemblyLine.Value), "", AssemblyLine.Value),
            MachineOrder, Switch(Machine.Value & "", "SB8", 1, "SB15", 2, 3),
            ShipYmd, If(IsBlank(ShipDate), "", Text(ShipDate, "yyyy-mm-dd")),
            PunchYmd, If(IsBlank(PunchDay), "", Text(PunchDay, "yyyy-mm-dd")),
            AsmYmd, If(IsBlank(AssemblyDate), "", Text(AssemblyDate, "yyyy-mm-dd")),
            ShipKey, If(IsBlank(ShipDate), 99999999, Year(ShipDate) * 10000 + Month(ShipDate) * 100 + Day(ShipDate)),
            PunchDayKey, If(IsBlank(PunchDay), 0, Year(PunchDay) * 10000 + Month(PunchDay) * 100 + Day(PunchDay)),

                (!Need12 || Done12) && (!Need14 || Done14) && (!Need16 || Done16) &&
                (!Need18 || Done18) && (!Need20 || Done20) && (!Need24 || Done24),
            NeedCount, If(Need12, 1, 0) + If(Need14, 1, 0) + If(Need16, 1, 0) + If(Need18, 1, 0) + If(Need20, 1, 0) + If(Need24, 1, 0),
            Gauges, Table(
                {G: 12, Need: Need12, Done: Done12, JobID: ID},
                {G: 14, Need: Need14, Done: Done14, JobID: ID},
                {G: 16, Need: Need16, Done: Done16, JobID: ID},

// ActiveJobs = ActiveBase + pair/group keys. GroupShipMaxKey (latest ship
// date in the pair) is the Turret tie-break after GroupShipKey, as machine.js:
// a clean 7/7 single sorts above a 7/7 + 7/8 pair.
// The With() record turns ActiveBase into an in-memory table ONCE per
// recalculation ("With ... create[s] collections internally", Learn delegation
// overview), so every per-row Filter/Min below runs locally, never as one
// server query per row. Only paired rows (small) are scanned per row.
ActiveJobs =
    With({base: ActiveBase},
    With({paired: Filter(base, !IsBlank(PairNo))},
    With({pairOrder: SortByColumns(paired, "MachineOrder", SortOrder.Ascending, "PunchDayKey", SortOrder.Ascending, "PunchSort", SortOrder.Ascending, "ID", SortOrder.Ascending)},
    With({ranked: ForAll(Sequence(CountRows(pairOrder)) As s, {PairNo: Index(pairOrder, s.Value).PairNo, Pos: s.Value})},
    With({firsts: Filter(ranked As r, r.Pos = Min(Filter(ranked, PairNo = r.PairNo), Pos))},
        AddColumns(
            AddColumns(
                base As j,
                GroupShipKey, If(IsBlank(j.PairNo), j.ShipKey, Min(Filter(paired, PairNo = j.PairNo), ShipKey)),
                GroupShipMaxKey, If(IsBlank(j.PairNo), j.ShipKey, Max(Filter(paired, PairNo = j.PairNo), If(ShipKey = 99999999, 0, ShipKey))),
                PunchGroupOrder, If(IsBlank(j.PairNo), j.PunchSort, Min(Filter(paired, PairNo = j.PairNo && PunchBand = j.PunchBand), PunchSort)),
                AsmGroupOrder, If(IsBlank(j.PairNo), j.AsmSort, Min(Filter(paired, PairNo = j.PairNo && AsmBand = j.AsmBand), AsmSort)),
                PairRank, If(IsBlank(j.PairNo), 0, With({fp: LookUp(firsts, PairNo = j.PairNo).Pos}, CountRows(Filter(firsts, Pos <= fp))))
            ),
            GroupShip, If(GroupShipKey = 99999999, Blank(), Date(RoundDown(GroupShipKey / 10000, 0), Mod(RoundDown(GroupShipKey / 100, 0), 100), Mod(GroupShipKey, 100))),
            PairColor, If(PairRank = 0, clrGrey, Index(clrPairs, Mod(PairRank - 1, 8) + 1).Value)
        )
    )))))
;

// ---------- Board helpers ----------
// Punch board day bands (Punch and TV). One row per band, in display order:
// the Unscheduled band (BandYmd = "") first when a shown job has no punch day,
// then today..today+5 (PunchWindow), plus every other day that holds a shown
// job, past or future (past days only while they hold one). Open = jobs not
// TurretDone; All = every active job ("Show completed" on).
// Built from the window plus the punch days in use (no continuous run of
// dates), so a job on a far-away punch day (for example a mistyped year)
// still gets its band; such a band's Label carries the year ("12/1/2027").
// Days are compared as yyyymmdd keys, so a date-only value that arrives with
// a time part can't drop a band.
PunchBandsOpen =
    With({src: Filter(ActiveJobs, !TurretDone && MachineText <> "")},
    With({
        hasUnscheduled: !IsEmpty(Filter(src, PunchDayKey = 0)),
        extra: Filter(Distinct(Filter(src, PunchDayKey > 0), PunchDayKey) As u, IsBlank(LookUp(PunchWindow, BandKey = u.Value)))
    },
        SortByColumns(
            Filter(
                ForAll(
                    Sequence(CountRows(PunchWindow) + CountRows(extra) + 1, 0) As s,
                    If(s.Value = 0,
                        {BandYmd: "", BandDate: Blank(), BandKey: 0, Label: "Unscheduled", IsToday: false},
                        With({k: If(s.Value <= CountRows(PunchWindow), Index(PunchWindow, s.Value).BandKey, Index(extra, s.Value - CountRows(PunchWindow)).Value)},
                        With({d: Date(RoundDown(k / 10000, 0), Mod(RoundDown(k / 100, 0), 100), Mod(k, 100))},
                            {BandYmd: Text(d, "yyyy-mm-dd"), BandDate: d, BandKey: k, Label: If(Abs(DateDiff(TodayDate, d, TimeUnit.Days)) > 400, Text(d, "m/d/yyyy", "en-US"), Text(d, "ddd mmm d", "en-US")), IsToday: k = TodayKey}
                        ))
                    )
                ),
                BandKey > 0 || hasUnscheduled
            ),
            "BandKey", SortOrder.Ascending
        )
    ))
;

PunchBandsAll =
    With({src: Filter(ActiveJobs, MachineText <> "")},
    With({
        hasUnscheduled: !IsEmpty(Filter(src, PunchDayKey = 0)),
        extra: Filter(Distinct(Filter(src, PunchDayKey > 0), PunchDayKey) As u, IsBlank(LookUp(PunchWindow, BandKey = u.Value)))
    },
        SortByColumns(
            Filter(
                ForAll(
                    Sequence(CountRows(PunchWindow) + CountRows(extra) + 1, 0) As s,
                    If(s.Value = 0,
                        {BandYmd: "", BandDate: Blank(), BandKey: 0, Label: "Unscheduled", IsToday: false},
                        With({k: If(s.Value <= CountRows(PunchWindow), Index(PunchWindow, s.Value).BandKey, Index(extra, s.Value - CountRows(PunchWindow)).Value)},
                        With({d: Date(RoundDown(k / 10000, 0), Mod(RoundDown(k / 100, 0), 100), Mod(k, 100))},
                            {BandYmd: Text(d, "yyyy-mm-dd"), BandDate: d, BandKey: k, Label: If(Abs(DateDiff(TodayDate, d, TimeUnit.Days)) > 400, Text(d, "m/d/yyyy", "en-US"), Text(d, "ddd mmm d", "en-US")), IsToday: k = TodayKey}
                        ))
                    )
                ),
                BandKey > 0 || hasUnscheduled
            ),
            "BandKey", SortOrder.Ascending
        )
    ))
;

// Assembly grid rows: Mon-Fri from the earlier of today and the earliest
// AssemblyDate of a SHOWN assigned job (AsmShown: unstarted, or started and
// not yet shipped), to the later of the latest AssemblyDate and the 10th
// workday from today. A Sat/Sun row appears only if a shown job is assigned to
// that day (so no shown job is ever without a row).
// AsmDaysAll: same over every assigned job ("Show started" on).
AsmDays =
    With({src: Filter(ActiveJobs, LineText <> "" && AsmYmd <> "" && AsmShown)},
    With({
        firstDay: First(SortByColumns(src, "AsmKey", SortOrder.Ascending)).AssemblyDate,
        lastDay: First(SortByColumns(src, "AsmKey", SortOrder.Descending)).AssemblyDate,
        usedKeys: Distinct(src, AsmKey),
        tenthWorkday: Last(FirstN(Filter(ForAll(Sequence(21, 0) As s, {D: DateAdd(TodayDate, s.Value, TimeUnit.Days)}), Weekday(D, StartOfWeek.Monday) <= 5), 10)).D
    },
    With({
        lo: If(IsBlank(firstDay) || firstDay >= TodayDate, TodayDate, firstDay < DateAdd(TodayDate, -400, TimeUnit.Days), DateAdd(TodayDate, -400, TimeUnit.Days), firstDay),
        hi: If(IsBlank(lastDay) || lastDay <= tenthWorkday, tenthWorkday, lastDay > DateAdd(TodayDate, 400, TimeUnit.Days), DateAdd(TodayDate, 400, TimeUnit.Days), lastDay)
    },
        Filter(
            ForAll(
                Sequence(DateDiff(lo, hi, TimeUnit.Days) + 1, 0) As s,
                With({d: DateAdd(lo, s.Value, TimeUnit.Days)},
                    {BandYmd: Text(d, "yyyy-mm-dd"), BandDate: d, BandKey: Year(d) * 10000 + Month(d) * 100 + Day(d), Label: Text(d, "ddd mmm d", "en-US"), IsToday: Text(d, "yyyy-mm-dd") = TodayYmd, IsWeekend: Weekday(d, StartOfWeek.Monday) > 5}
                )
            ),
            !IsWeekend || BandKey in usedKeys
        )
    )))
;
